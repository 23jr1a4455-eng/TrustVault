# TrustVault: database trigger that refuses to change or delete audit rows (PostgreSQL only).
#
# HOW TO USE
#   1. python manage.py makemigrations vault --empty --name audit_append_only_trigger
#   2. Open the new file in vault\migrations\ (the highest number).
#   3. Replace its WHOLE content with the code below.
#   4. Change PREVIOUS_MIGRATION to the name of the migration just before it (without .py),
#      for example "0004_auditchainhead_auditevent_prev_hash_and_more".
#      The empty file Django generated already shows that name in its dependencies line.

from django.db import migrations

FORWARDS = """
CREATE OR REPLACE FUNCTION vault_audit_block_change() RETURNS trigger AS $$
BEGIN
    RAISE EXCEPTION 'vault_auditevent is append-only'
        USING ERRCODE = 'insufficient_privilege';
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS vault_audit_append_only ON vault_auditevent;
CREATE TRIGGER vault_audit_append_only
    BEFORE UPDATE OR DELETE ON vault_auditevent
    FOR EACH ROW EXECUTE FUNCTION vault_audit_block_change();
"""

BACKWARDS = """
DROP TRIGGER IF EXISTS vault_audit_append_only ON vault_auditevent;
DROP FUNCTION IF EXISTS vault_audit_block_change();
"""


def install(apps, schema_editor):
    if schema_editor.connection.vendor == "postgresql":
        schema_editor.execute(FORWARDS, params=None)


def uninstall(apps, schema_editor):
    if schema_editor.connection.vendor == "postgresql":
        schema_editor.execute(BACKWARDS, params=None)
